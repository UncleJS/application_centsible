# Centsible Pod
# All containers share localhost networking within the pod.
# Web: 10300, API: 10301, MariaDB: internal only (3306)

[Unit]
Description=Centsible Pod
Documentation=https://docs.podman.io/en/latest/markdown/podman-pod-create.1.html

[Pod]
PodName=centsible
Network=slirp4netns:port_handler=slirp4netns
# Bind to loopback only — local-only deployment. To expose externally,
# put a reverse proxy on the host that listens on 0.0.0.0 and forwards
# to 127.0.0.1:10300 / 127.0.0.1:10301.
PublishPort=127.0.0.1:10300:10300
PublishPort=127.0.0.1:10301:10301
# 3306 — MariaDB NOT exposed externally

[Install]
WantedBy=default.target
